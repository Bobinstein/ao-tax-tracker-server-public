# NGINX CONFIGURATION UPDATE FOR LONG-RUNNING PDF GENERATION
# ============================================================
#
# Update your nginx site config, e.g.:
#   /etc/nginx/sites-available/YOUR_SERVER_HOSTNAME
#
# Change the timeout settings from 300s to 600s (10 minutes):
#
# BEFORE:
#     proxy_connect_timeout 300s;
#     proxy_send_timeout 300s;
#     proxy_read_timeout 300s;
#     send_timeout 300s;
#
# AFTER:
#     proxy_connect_timeout 600s;
#     proxy_send_timeout 600s;
#     proxy_read_timeout 600s;
#     send_timeout 600s;
#
# Run these commands to update (replace YOUR_SERVER_HOSTNAME):
# ============================================================

SITE=/etc/nginx/sites-available/YOUR_SERVER_HOSTNAME

# 1. Edit the nginx config:
sudo nano "$SITE"

# 2. Replace the timeout values (or use sed):
sudo sed -i 's/proxy_connect_timeout 300s/proxy_connect_timeout 600s/g' "$SITE"
sudo sed -i 's/proxy_send_timeout 300s/proxy_send_timeout 600s/g' "$SITE"
sudo sed -i 's/proxy_read_timeout 300s/proxy_read_timeout 600s/g' "$SITE"
sudo sed -i 's/send_timeout 300s/send_timeout 600s/g' "$SITE"

# 3. Test the nginx configuration:
sudo nginx -t

# 4. If the test passes, reload nginx:
sudo systemctl reload nginx

# 5. Verify the changes:
grep -E "timeout" "$SITE"

# ============================================================
# ALSO: Restart the Node.js server to apply the new timeout settings
# ============================================================

pm2 restart ao-tax-tracker-server

sudo systemctl restart ao-tax-tracker-server
